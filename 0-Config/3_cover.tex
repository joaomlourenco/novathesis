%-------------------------------------------------------------
% Advisers' names
%
%     \ntaddperson{adviser}(<role>,<gender>){Full Name}
%       [title=..., position=..., institution=...]
%.............................................................
%
%   <role>: a  -> adviser
%           c  -> co-adviser
%           co -> co-adviser (same as 'c')
%           t  -> tutor
%
% <gender>: m -> male
%           f -> female
%
% title, position, and institution are all optional -- omit any (or all)
% of them, or the whole bracket, if not needed.
%
% Legacy syntax (still accepted, but deprecated -- only the first two
% commas are significant, so a comma-containing institution stays whole):
%     \ntaddperson{adviser}(<role>,<gender>){Full Name, Position, and School}
%
% You may use "\\" to force line breaks
%-------------------------------------------------------------
% other/huberlin's cover format ("<akad. Grade/Vorname/Name>") expects an
% academic-degree title prefixed to the name; every other school's demo
% preview stays exactly as it was (no title). The [...] bracket is read
% unexpanded by \ntaddperson (it's passed straight into \keys_set:nn), so
% a macro placeholder for "title=...," would never expand there -- the
% two branches below spell out the full bracket content literally instead.
\ifoptionequal{/novathesis/schoolgiven}{other/huberlin}{%
  % Advisers(s)
  \ntaddperson{adviser}(a,f){Mary Doe Adviser Name}
    [title=Prof.\ Dr., position=Full Professor, institution=NOVA University Lisbon]
  % Co-advisers(s)
  \ntaddperson{adviser}(c,m){John Doe Co-Adviser Name}
    [title=Prof.\ Dr., position=Associate Professor, institution=NOVA University Lisbon]
  \ntaddperson{adviser}(c,f){John Doe other Co-Adviser Name}
    [title=Prof.\ Dr., position=Full Professor, institution=NOVA University Lisbon]
}{%
  % Advisers(s)
  \ntaddperson{adviser}(a,f){Mary Doe Adviser Name}
    [position=Full Professor, institution=NOVA University Lisbon]
  % Co-advisers(s)
  \ntaddperson{adviser}(c,m){John Doe Co-Adviser Name}
    [position=Associate Professor, institution=NOVA University Lisbon]
  \ntaddperson{adviser}(c,f){John Doe other Co-Adviser Name}
    [position=Full Professor, institution=NOVA University Lisbon]
}%
